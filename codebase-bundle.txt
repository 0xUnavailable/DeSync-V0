// Server exposes read + write endpoints for the DeSync node.
type Server struct {
	Node      *network.Node
	Store     *state.MemStore
	BlockDir  string
	Addr      string
	StaticDir string
}

func NewServer(node *network.Node, store *state.MemStore, blockDir, addr, staticDir string) *Server {
	return &Server{
		Node:      node,
		Store:     store,
		BlockDir:  blockDir,
		Addr:      addr,
		StaticDir: staticDir,
	}
}

// Start starts the HTTP server (blocking).
func (s *Server) Start() {
	// API endpoints
	http.HandleFunc("/status", s.handleStatus)
	http.HandleFunc("/state/land/", s.handleLand)
	http.HandleFunc("/state/player/", s.handlePlayer)
	http.HandleFunc("/state/world", s.handleWorld) // ✅ new endpoint
	http.HandleFunc("/lastRoot", s.handleLastRoot)
	http.HandleFunc("/blocks/", s.handleBlock)
	http.HandleFunc("/blocks/latest", s.handleLast)
	http.HandleFunc("/months/", s.handleMonthSummary)
	http.HandleFunc("/months/latest", s.handleLatestMonth)
	http.HandleFunc("/tx", s.handleTx)
	http.HandleFunc("/txs/all", s.handleAllTxs)
	http.HandleFunc("/txs/block/", s.handleBlockTxs)

	// Serve static files (explorer UI)
	if s.StaticDir != "" {
		fs := http.FileServer(http.Dir(s.StaticDir))
		http.Handle("/", fs)
	}

	srv := &http.Server{
		Addr:         s.Addr,
		ReadTimeout:  10 * time.Second,
		WriteTimeout: 20 * time.Second,
	}

	log.Printf("API server listening on %s", s.Addr)
	if err := srv.ListenAndServe(); err != nil && err != http.ErrServerClosed {
		log.Fatalf("API server failed: %v", err)
	}
}

func enableCORS(w http.ResponseWriter) {
	w.Header().Set("Access-Control-Allow-Origin", "*")
	w.Header().Set("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
	w.Header().Set("Access-Control-Allow-Headers", "Content-Type")
}

func handlePreflight(w http.ResponseWriter, r *http.Request) bool {
	enableCORS(w)
	if r.Method == http.MethodOptions {
		w.WriteHeader(http.StatusNoContent)
		return true
	}
	return false
}

func serveFileOr404(w http.ResponseWriter, path string) {
	enableCORS(w)
	data, err := os.ReadFile(path)
	if err != nil {
		http.Error(w, "not found", http.StatusNotFound)
		return
	}
	w.Header().Set("Content-Type", "application/json")
	_, _ = w.Write(data)
}

func bytesToStr(b []byte) string {
	if len(b) == 0 {
		return ""
	}
	return string(b)
}

func writeJSON(w http.ResponseWriter, v interface{}) {
	enableCORS(w)
	w.Header().Set("Content-Type", "application/json")
	_ = json.NewEncoder(w).Encode(v)
}
func computeTxID(tx *types.Transaction) string {
	h := sha256.New()
	h.Write([]byte(tx.Sender))
	h.Write([]byte(fmt.Sprintf("%d", tx.Nonce)))
	h.Write([]byte(tx.Action))
	if tx.Payload != nil {
		keys := make([]string, 0, len(tx.Payload))
		for k := range tx.Payload {
			keys = append(keys, k)
		}
		sort.Strings(keys)
		for _, k := range keys {
			h.Write([]byte(k))
			h.Write([]byte(tx.Payload[k]))
		}
	}
	if len(tx.PubKey) > 0 {
		h.Write([]byte(tx.PubKey))  // Convert string to []byte
	}
	if len(tx.Sig) > 0 {
		h.Write([]byte(tx.Sig))  // Convert string to []byte
	}
	return hex.EncodeToString(h.Sum(nil))
}

func (s *Server) handleStatus(w http.ResponseWriter, r *http.Request) {
	if handlePreflight(w, r) {
		return
	}

	var peerID string
	var nodeRole string
	var isReady bool

	if s.Node != nil && s.Node.Host != nil {
		peerID = s.Node.Host.ID().String()
		isReady = true
		if s.Node.TxTopic != nil {
			nodeRole = "leader"
		} else {
			nodeRole = "worker"
		}
	} else {
		peerID = "unknown"
		nodeRole = "unknown"
		isReady = false
	}

	stateSize := 0
	if s.Store != nil {
		snap := s.Store.GetSnapshot()
		stateSize = len(snap)
	}

	blockCount := 0
	if s.BlockDir != "" {
		if entries, err := os.ReadDir(s.BlockDir); err == nil {
			for _, e := range entries {
				if strings.HasPrefix(e.Name(), "block-") && strings.HasSuffix(e.Name(), ".json") {
					blockCount++
				}
			}
		}
	}

	resp := map[string]interface{}{
		"peer_id":     peerID,
		"role":        nodeRole,
		"ready":       isReady,
		"state_keys":  stateSize,
		"block_count": blockCount,
		"timestamp":   time.Now().Unix(),
		"api_version": "1.0",
	}

	writeJSON(w, resp)
}

// canonicalTxBytes produces deterministic bytes for signing/verifying.
// It mirrors the JS canonicalize for fields: sender, nonce, action, payload (map[string]string)
func canonicalTxBytes(sender string, nonce uint64, action string, payload map[string]string) ([]byte, error) {
    // Match JavaScript canonicalize exactly
    // Order: action, nonce, payload, sender (alphabetical)
    
    var parts []string
    
    // action
    actionJSON, _ := json.Marshal(action)
    parts = append(parts, `"action":`+string(actionJSON))
    
    // nonce
    parts = append(parts, fmt.Sprintf(`"nonce":%d`, nonce))
    
    // payload with sorted keys
    if len(payload) > 0 {
        keys := make([]string, 0, len(payload))
        for k := range payload {
            keys = append(keys, k)
        }
        sort.Strings(keys)
        
        var payloadParts []string
        for _, k := range keys {
            keyJSON, _ := json.Marshal(k)
            valJSON, _ := json.Marshal(payload[k])
            payloadParts = append(payloadParts, string(keyJSON)+":"+string(valJSON))
        }
        parts = append(parts, `"payload":{`+strings.Join(payloadParts, ",")+`}`)
    } else {
        parts = append(parts, `"payload":{}`)
    }
    
    // sender
    senderJSON, _ := json.Marshal(sender)
    parts = append(parts, `"sender":`+string(senderJSON))
    
    canonical := "{" + strings.Join(parts, ",") + "}"
    return []byte(canonical), nil
}
// verifyTxSignature verifies base64 SPKI public key and base64 signature against canonical tx bytes.
// Returns nil on success.
func verifyTxSignature(txSender string, nonce uint64, action string, payload map[string]string, pubKeyB64 string, sigB64 string) error {
    if pubKeyB64 == "" || sigB64 == "" {
        return errors.New("missing pubkey or signature")
    }
    
    // decode pubkey (SPKI, DER)
    pubBytes, err := base64.StdEncoding.DecodeString(pubKeyB64)
    if err != nil {
        return fmt.Errorf("pubkey base64 decode: %w", err)
    }
    
    // parse spki
    keyIface, err := x509.ParsePKIXPublicKey(pubBytes)
    if err != nil {
        return fmt.Errorf("parse pubkey spki failed: %w", err)
    }
    ecdsaPub, ok := keyIface.(*ecdsa.PublicKey)
    if !ok {
        return fmt.Errorf("pubkey not ecdsa")
    }
    
    // Build canonical message bytes
    msg, err := canonicalTxBytes(txSender, nonce, action, payload)
    if err != nil {
        return fmt.Errorf("canonicalize failed: %w", err)
    }
    
    h := sha256.Sum256(msg)
    
    // decode signature
    sigBytes, err := base64.StdEncoding.DecodeString(sigB64)
    if err != nil {
        return fmt.Errorf("sig base64 decode: %w", err)
    }
    
    // verify - JavaScript produces IEEE P1363 format (r||s, 64 bytes for P-256)
    if len(sigBytes) != 64 {
        return fmt.Errorf("expected 64-byte P1363 signature, got %d bytes", len(sigBytes))
    }

    // Split into r and s (32 bytes each for P-256)
    r := new(big.Int).SetBytes(sigBytes[:32])
    s := new(big.Int).SetBytes(sigBytes[32:])

    if !ecdsa.Verify(ecdsaPub, h[:], r, s) {
        return fmt.Errorf("signature verification failed")
    }
    
    // verify that txSender matches hash of pubkey (identity binding)
    pubHash := sha256.Sum256(pubBytes)
    expectedSender := hex.EncodeToString(pubHash[:])
    if txSender != expectedSender {
        return fmt.Errorf("sender mismatch: tx.sender=%s computed=%s", txSender, expectedSender)
    }
    
    return nil
}

// ✅ New handler for full world snapshot
func (s *Server) handleWorld(w http.ResponseWriter, r *http.Request) {
	if handlePreflight(w, r) {
		return
	}

	snap := s.Store.GetSnapshot()

	type plot struct {
		ID       string `json:"id"`
		Owner    string `json:"owner"`
		Resource int    `json:"resource"`
		Quota    int    `json:"quota"`
	}

	seen := map[string]struct{}{}
	for k := range snap {
		if strings.HasPrefix(k, "land:") {
			parts := strings.Split(k, ":")
			if len(parts) >= 3 {
				seen[parts[1]] = struct{}{}
			}
		}
	}

	plots := make([]plot, 0, len(seen))
	for pid := range seen {
		owner := ""
		if b, ok := s.Store.Get([]byte(fmt.Sprintf("land:%s:owner", pid))); ok && len(b) > 0 {
			owner = string(b)
		}
		resource := 0
		if b, ok := snap[fmt.Sprintf("land:%s:resource", pid)]; ok && len(b) > 0 {
			if n, err := strconv.Atoi(string(b)); err == nil {
				resource = n
			}
		}
		quota := 0
		if b, ok := snap[fmt.Sprintf("land:%s:quota", pid)]; ok && len(b) > 0 {
			if n, err := strconv.Atoi(string(b)); err == nil {
				quota = n
			}
		}
		plots = append(plots, plot{
			ID:       pid,
			Owner:    owner,
			Resource: resource,
			Quota:    quota,
		})
	}

	writeJSON(w, plots)
}

func (s *Server) handleBlock(w http.ResponseWriter, r *http.Request) {
	if handlePreflight(w, r) {
		return
	}
	trim := strings.TrimPrefix(r.URL.Path, "/blocks/")
	if trim == "" {
		http.Error(w, "missing block id", http.StatusBadRequest)
		return
	}
	if trim == "latest" {
		s.handleLast(w, r)
		return
	}
	n, err := strconv.Atoi(trim)
	if err == nil {
		filename := fmt.Sprintf("block-%06d.json", n)
		path := filepath.Join(s.BlockDir, filename)
		serveFileOr404(w, path)
		return
	}
	path := filepath.Join(s.BlockDir, fmt.Sprintf("block-%s.json", trim))
	serveFileOr404(w, path)
}

func (s *Server) handleLast(w http.ResponseWriter, r *http.Request) {
	if handlePreflight(w, r) {
		return
	}
	entries, err := os.ReadDir(s.BlockDir)
	if err != nil || len(entries) == 0 {
		http.Error(w, "no blocks available", http.StatusNotFound)
		return
	}
	max := -1
	var latestName string
	for _, e := range entries {
		name := e.Name()
		if !strings.HasPrefix(name, "block-") || !strings.HasSuffix(name, ".json") {
			continue
		}
		numStr := strings.TrimSuffix(strings.TrimPrefix(name, "block-"), ".json")
		n, err := strconv.Atoi(numStr)
		if err != nil {
			continue
		}
		if n > max {
			max = n
			latestName = name
		}
	}
	if latestName == "" {
		http.Error(w, "no blocks found", http.StatusNotFound)
		return
	}
	path := filepath.Join(s.BlockDir, latestName)
	serveFileOr404(w, path)
}

func (s *Server) handleTx(w http.ResponseWriter, r *http.Request) {
	if handlePreflight(w, r) {
		return
	}
	if r.Method != http.MethodPost {
		http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
		return
	}

	body, err := io.ReadAll(r.Body)
	if err != nil {
		http.Error(w, "failed to read body", http.StatusBadRequest)
		return
	}

	var tx types.Transaction
	if err := json.Unmarshal(body, &tx); err != nil {
		http.Error(w, "invalid transaction JSON", http.StatusBadRequest)
		return
	}

	// 🔐 Verify signature if PubKey & Sig are present
	if len(tx.PubKey) == 0 || len(tx.Sig) == 0 {
		http.Error(w, "missing pubkey or signature", http.StatusBadRequest)
		return
	}

	if err := verifyTxSignature(tx.Sender, tx.Nonce, tx.Action, tx.Payload, tx.PubKey, tx.Sig); err != nil {
		http.Error(w, "invalid signature: "+err.Error(), http.StatusBadRequest)
		return
	}

	if strings.TrimSpace(tx.ID) == "" {
		tx.ID = computeTxID(&tx)
	}

	msg := network.TxMessage{Tx: &tx}
	data, err := network.EncodeTx(msg)
	if err != nil {
		http.Error(w, "failed to encode tx", http.StatusInternalServerError)
		return
	}

	if s.Node == nil || s.Node.TxTopic == nil {
		http.Error(w, "node tx topic not available on server", http.StatusInternalServerError)
		return
	}

	if err := s.Node.TxTopic.Publish(context.Background(), data); err != nil {
		http.Error(w, fmt.Sprintf("failed to publish tx: %v", err), http.StatusInternalServerError)
		return
	}

	resp := map[string]string{"status": "submitted", "txid": tx.ID}
	writeJSON(w, resp)
}

func (s *Server) handleAllTxs(w http.ResponseWriter, r *http.Request) {
	if handlePreflight(w, r) {
		return
	}
	entries, err := os.ReadDir(s.BlockDir)
	if err != nil {
		http.Error(w, "failed to read blocks dir", http.StatusInternalServerError)
		return
	}

	var allTxs []map[string]interface{}
	for _, e := range entries {
		if !strings.HasPrefix(e.Name(), "block-") || !strings.HasSuffix(e.Name(), ".json") {
			continue
		}
		path := filepath.Join(s.BlockDir, e.Name())
		data, err := os.ReadFile(path)
		if err != nil {
			continue
		}
		var block struct {
			Txs []types.Transaction `json:"txs"`
		}
		if err := json.Unmarshal(data, &block); err == nil {
			for _, tx := range block.Txs {
				txMap := map[string]interface{}{
					"id":      tx.ID,
					"sender":  tx.Sender,
					"nonce":   tx.Nonce,
					"action":  tx.Action,
					"payload": tx.Payload,
				}
				allTxs = append(allTxs, txMap)
			}
		}
	}

	writeJSON(w, allTxs)
}

// Leader is a single static leader (no rotation).
type Leader struct {
	ID      string
	NetNode *network.Node

	// authoritative state
	State *state.MemStore

	// tx queue
	txQueueMu sync.Mutex
	txQueue   []*types.Transaction

	// pending batches
	pendingMu           sync.Mutex
	pendingBatchIDs     []string
	pendingBatchTxs     map[string][]*types.Transaction
	nextBatchSequential int

	// simple block height
	blockMu sync.Mutex
	height  int
}

func NewLeader(id string) *Leader {
	return &Leader{
		ID:                  id,
		State:               state.NewMemStore(),
		txQueue:             make([]*types.Transaction, 0, 1024),
		pendingBatchTxs:     make(map[string][]*types.Transaction),
		pendingBatchIDs:     make([]string, 0),

	}
}

// InitWorld seeds the in-memory state with a size x size grid of plots.
// It is idempotent: it will not overwrite existing owner/resource/quota keys.
func (l *Leader) InitWorld(size int) {
	if size <= 0 {
		return
	}
	
	// Initialize genesis account - use the hash from your loadgen output
	genesisAccount := "842b4e0a5f891d7ddd612bbed4e006b30029777618089e11e4434b23aa8f436e"
	genesisKey := fmt.Sprintf("player:%s:balance", genesisAccount)
	l.State.Set([]byte(genesisKey), []byte("1000000000000000000000"))
	log.Printf("Leader: initialized genesis account %s with 10000M balance", genesisAccount)
	
	for x := 0; x < size; x++ {
		for y := 0; y < size; y++ {
			plotID := fmt.Sprintf("plot-%d-%d", x, y)
			ownerKey := fmt.Sprintf("land:%s:owner", plotID)
			if b, ok := l.State.Get([]byte(ownerKey)); !ok || len(b) == 0 {
				l.State.Set([]byte(ownerKey), []byte(""))
			}
			quotaKey := fmt.Sprintf("land:%s:quota", plotID)
			if b, ok := l.State.Get([]byte(quotaKey)); !ok || len(b) == 0 {
				l.State.Set([]byte(quotaKey), []byte("0"))
			}
			resKey := fmt.Sprintf("land:%s:resource", plotID)
			if b, ok := l.State.Get([]byte(resKey)); !ok || len(b) == 0 {
				l.State.Set([]byte(resKey), []byte("0"))
			}
		}
	}
	log.Printf("Leader %s: initialized world grid %dx%d", l.ID, size, size)
}
// === Worker registry ===
func (l *Leader) StartWorkerRegistry(ctx context.Context) error {
	sub, err := l.NetNode.WorkersTopic.Subscribe()
	if err != nil {
		return err
	}
	go func() {
		for {
			msg, err := sub.Next(ctx)
			if err != nil {
				return
			}
			wh, err := network.DecodeWorkerHello(msg.Data)
			if err != nil {
				continue
			}
			log.Printf("Leader %s: observed worker %s (peer=%s)", l.ID, wh.WorkerID, wh.PeerID)
		}
	}()
	return nil
}

func (l *Leader) StartTxListener(ctx context.Context) error {

	if err != nil {

	}
	log.Printf("Leader %s: subscribed to TxTopic", l.ID)

	go func() {
		for {
			msg, err := sub.Next(ctx)
			if err != nil {
				return
			}
			txm, err := network.DecodeTx(msg.Data)
			if err != nil {
				continue
			}
			tx := txm.Tx
			if tx == nil {
				continue
			}
			// signature check delegated to workers; leader accepts valid formatted tx
			l.enqueueTx(tx)
		}
	}()
	return nil
}

func (l *Leader) enqueueTx(tx *types.Transaction) {
	l.txQueueMu.Lock()
	l.txQueue = append(l.txQueue, tx)
	l.txQueueMu.Unlock()
}

func (l *Leader) dequeueBatch(max int) []*types.Transaction {
	l.txQueueMu.Lock()
	defer l.txQueueMu.Unlock()
	if len(l.txQueue) == 0 {
		return nil
	}
	n := max
	if len(l.txQueue) < n {
		n = len(l.txQueue)
	}
	batch := l.txQueue[:n]
	l.txQueue = l.txQueue[n:]
	return batch
}

// === Batch loop ===
func (l *Leader) RunBatchLoop(ctx context.Context, batchSize int, interval time.Duration) {
	ticker := time.NewTicker(interval)
	defer ticker.Stop()
	log.Printf("Leader %s: starting batch loop (batchSize=%d interval=%s)", l.ID, batchSize, interval)

	for {
		select {
		case <-ctx.Done():
			return
		case <-ticker.C:
			txs := l.dequeueBatch(batchSize)
			if len(txs) == 0 {
				continue
			}
			batchID := fmt.Sprintf("batch-%d", l.nextBatchSequential)
			l.nextBatchSequential++

			l.pendingMu.Lock()
			l.pendingBatchTxs[batchID] = txs
			l.pendingBatchIDs = append(l.pendingBatchIDs, batchID)
			l.pendingMu.Unlock()

			// include authoritative snapshot
			snapshot := l.State.GetSnapshot()

			bm := network.BatchMessage{
				BatchID:      batchID,
				Txs:          txs,
				ShardIndex:   0,
				ShardCount:   1,
				TargetWorker: "",
				Snapshot:     snapshot,
			}
			data, _ := network.Encode(bm)
			if err := l.NetNode.BatchTopic.Publish(ctx, data); err != nil {
				log.Printf("Leader %s: publish batch error: %v", l.ID, err)
			} else {
				log.Printf("Leader %s: published batch %s (txs=%d snapshot_keys=%d)", l.ID, batchID, len(txs), len(snapshot))
			}
		}
	}
}

// === Results collector ===
func (l *Leader) ListenResults(ctx context.Context) error {
	sub, err := l.NetNode.ResultTopic.Subscribe()
	if err != nil {
		return err
	}

	go func() {
		for {
			msg, err := sub.Next(ctx)
			if err != nil {
				return
			}
			sr, err := network.DecodeResult(msg.Data)
			if err != nil {
				continue
			}

			applied := 0
			failed := 0
			for _, r := range sr.Results {
				if r == nil {
					continue
				}
				if r.Success {
					for k, v := range r.WriteSet {
						l.State.Set([]byte(k), v)
						applied++
					}
				} else {
					failed++
				}
			}

			log.Printf("Leader %s: results from worker=%s batch=%s applied_writes=%d failures=%d",
				l.ID, sr.WorkerID, sr.BatchID, applied, failed)
		}
	}()

	return nil
}

// === Block ticker and month enforcement ===
// BlockTicker advances height, commits blocks, and runs month-end enforcement when appropriate.
func (l *Leader) BlockTicker(ctx context.Context, interval time.Duration, monthBlockCount int) {
	ticker := time.NewTicker(interval)
	defer ticker.Stop()

	blocksDir := "blocks"
	_ = os.MkdirAll(blocksDir, 0755)
	indexFile := filepath.Join(blocksDir, "index.json")

	// load existing index if any
	var index []map[string]interface{}
	if data, err := os.ReadFile(indexFile); err == nil {
		_ = json.Unmarshal(data, &index)
	}

	log.Printf("Leader %s: starting block ticker (interval=%s monthBlocks=%d)", l.ID, interval.String(), monthBlockCount)

	for {
		select {
		case <-ctx.Done():
			return
		case <-ticker.C:
			// advance height
			l.blockMu.Lock()
			l.height++
			height := l.height
			l.blockMu.Unlock()

			// grab pending
			l.pendingMu.Lock()
			pendingIDs := l.pendingBatchIDs
			batchesMap := l.pendingBatchTxs
			l.pendingBatchIDs = nil
			l.pendingBatchTxs = make(map[string][]*types.Transaction)
			l.pendingMu.Unlock()

			collected := make([]*types.Transaction, 0)
			for _, bid := range pendingIDs {
				if txs, ok := batchesMap[bid]; ok {
					collected = append(collected, txs...)
				}
			}

			// commit block if there are txs
			if len(collected) > 0 {
				blk := types.Block{
					Height: height,
					Txs:    collected,
					Root:   []byte("dummy-root"),
					Time:   time.Now(),
				}

				filename := fmt.Sprintf("block-%06d.json", blk.Height)
				fullPath := filepath.Join(blocksDir, filename)
				data, _ := json.MarshalIndent(blk, "", "  ")
				if err := os.WriteFile(fullPath, data, 0644); err != nil {
					log.Printf("Leader %s: failed to save block #%d: %v", l.ID, blk.Height, err)
				} else {
					// append to index
					entry := map[string]interface{}{
						"height":   blk.Height,
						"filename": filename,
						"txs":      len(collected),
						"time":     blk.Time,
					}
					index = append(index, entry)
					indexData, _ := json.MarshalIndent(index, "", "  ")
					_ = os.WriteFile(indexFile, indexData, 0644)

					log.Printf("Leader %s: committed block #%d (txs=%d)", l.ID, blk.Height, len(collected))
				}
			}

			// month-end check
			if monthBlockCount > 0 && (height%monthBlockCount) == 0 {
				monthNum := height / monthBlockCount
				log.Printf("Leader %s: month #%d boundary reached (height=%d) — running tax enforcement", l.ID, monthNum, height)
				if err := l.runMonthEndEnforcement(monthNum); err != nil {
					log.Printf("Leader %s: month enforcement error: %v", l.ID, err)
				}
			}
		}
	}
}

// runMonthEndEnforcement iterates over land entries and enforces tax rules.
// Writes month-summary-<n>.json into blocks/.
func (l *Leader) runMonthEndEnforcement(month int) error {
	// Configurable rules (tweak these)
	const taxRate = 1.0 / 10.0   // 10%
	const penaltyRate = 1.0 / 5.0 // 20%

	// snapshot copy
	snap := l.State.GetSnapshot()

	type summaryEntry struct {
		PlotID     string `json:"plot_id"`
		Owner      string `json:"owner"`
		Resource   int    `json:"resource"`
		TaxDue     int    `json:"tax_due"`
		TaxPaid    bool   `json:"tax_paid"`
		Penalty    int    `json:"penalty_applied"`
		OwnerAfter int    `json:"owner_balance_after"`
		Note       string `json:"note,omitempty"`
	}

	summary := make([]summaryEntry, 0)

	// helpers
	readIntFromSnap := func(key string) int {
		if v, ok := snap[key]; ok && len(v) > 0 {
			if n, err := strconv.Atoi(string(v)); err == nil {
				return n
			}
		}
		return 0
	}

	applySet := func(k string, v []byte) {
		l.State.Set([]byte(k), v)
	}

	// collect plot IDs by scanning snapshot keys
	seenPlots := map[string]struct{}{}
	for k := range snap {
		if strings.HasPrefix(k, "land:") {
			parts := strings.Split(k, ":")
			if len(parts) >= 3 {
				plotID := parts[1]
				seenPlots[plotID] = struct{}{}
			}
		}
	}

	for plotID := range seenPlots {
		ownerKey := fmt.Sprintf("land:%s:owner", plotID)
		owner := ""
		if b, ok := l.State.Get([]byte(ownerKey)); ok && len(b) > 0 {
			owner = string(b)
		}
		resource := readIntFromSnap(fmt.Sprintf("land:%s:resource", plotID))
		quota := readIntFromSnap(fmt.Sprintf("land:%s:quota", plotID))

		var taxDue int
		if quota > 0 {
			taxDue = quota
		} else {
			taxDue = int(float64(resource) * taxRate)
		}
		if taxDue < 0 {
			taxDue = 0
		}

		entry := summaryEntry{
			PlotID:   plotID,
			Owner:    owner,
			Resource: resource,
			TaxDue:   taxDue,
		}

		if owner == "" {
			entry.Note = "unowned plot"
			summary = append(summary, entry)
			continue
		}

		ownerBalKey := fmt.Sprintf("player:%s:balance", owner)
		ownerBal := 0
		if b, ok := l.State.Get([]byte(ownerBalKey)); ok && len(b) > 0 {
			if n, err := strconv.Atoi(string(b)); err == nil {
				ownerBal = n
			}
		}

		if ownerBal >= taxDue {
			ownerBalAfter := ownerBal - taxDue
			applySet(ownerBalKey, []byte(strconv.Itoa(ownerBalAfter)))
			applySet(fmt.Sprintf("land:%s:tax_paid", plotID), []byte("true"))
			entry.TaxPaid = true
			entry.OwnerAfter = ownerBalAfter
		} else {
			penalty := int(float64(resource) * penaltyRate)
			if penalty < 0 {
				penalty = 0
			}
			newResource := resource - penalty
			if newResource < 0 {
				newResource = 0
			}
			applySet(fmt.Sprintf("land:%s:resource", plotID), []byte(strconv.Itoa(newResource)))
			applySet(fmt.Sprintf("land:%s:tax_paid", plotID), []byte("false"))
			entry.TaxPaid = false
			entry.Penalty = penalty
			entry.OwnerAfter = ownerBal
			entry.Note = "insufficient balance; penalty applied"
		}

		summary = append(summary, entry)
	}

	// write summary
		if len(summary) > 0 {
			blocksDir := "blocks"
			_ = os.MkdirAll(blocksDir, 0755)
			filename := fmt.Sprintf("month-summary-%06d.json", month)
			fullPath := filepath.Join(blocksDir, filename)
			if data, err := json.MarshalIndent(summary, "", "  "); err == nil {
				_ = os.WriteFile(fullPath, data, 0644)
			} else {
				log.Printf("Leader %s: failed to marshal month summary: %v", l.ID, err)
			}
			log.Printf("Leader %s: month %d enforcement complete — plots=%d", l.ID, month, len(summary))
		} else {
			log.Printf("Leader %s: month %d enforcement skipped (no activity)", l.ID, month)
		}
		return nil
}

	PubKey  string            `json:"PubKey"`
	Sig     string            `json:"Sig"`
}

// canonicalize produces the same JSON format as JavaScript frontend
func canonicalize(sender string, nonce uint64, action string, payload map[string]string) string {
	// Sort payload keys
	keys := make([]string, 0, len(payload))
	for k := range payload {
		keys = append(keys, k)
	}
	sort.Strings(keys)

	// Build JSON with alphabetical field order: action, nonce, payload, sender
	var parts []string

	actionJSON, _ := json.Marshal(action)
	parts = append(parts, `"action":`+string(actionJSON))

	parts = append(parts, fmt.Sprintf(`"nonce":%d`, nonce))

	if len(payload) > 0 {
		var payloadParts []string
		for _, k := range keys {
			keyJSON, _ := json.Marshal(k)
			valJSON, _ := json.Marshal(payload[k])
			payloadParts = append(payloadParts, string(keyJSON)+":"+string(valJSON))
		}
		parts = append(parts, `"payload":{`+strings.Join(payloadParts, ",")+`}`)
	} else {
		parts = append(parts, `"payload":{}`)
	}

	senderJSON, _ := json.Marshal(sender)
	parts = append(parts, `"sender":`+string(senderJSON))

	return "{" + strings.Join(parts, ",") + "}"
}

// sendTxn builds, signs with ECDSA P-256, and POSTs a transaction
func sendTxn(ctx context.Context, priv *ecdsa.PrivateKey, sender, action string, payload map[string]string, api string) string {
	nonce := uint64(time.Now().UnixNano())

	// Canonicalize for signing
	canonical := canonicalize(sender, nonce, action, payload)
	hash := sha256.Sum256([]byte(canonical))

	// Sign with ECDSA
	r, s, err := ecdsa.Sign(rand.Reader, priv, hash[:])
	if err != nil {
		log.Printf("sign error: %v", err)
		return ""
	}

	// Convert signature to P1363 format (r||s, 32 bytes each for P-256)
	sigBytes := make([]byte, 64)
	r.FillBytes(sigBytes[0:32])
	s.FillBytes(sigBytes[32:64])

	// Export public key as SPKI
	pubBytes, err := x509.MarshalPKIXPublicKey(&priv.PublicKey)
	if err != nil {
		log.Printf("marshal pubkey error: %v", err)
		return ""
	}

	tx := Transaction{
		ID:      "",
		Sender:  sender,
		Nonce:   nonce,
		Action:  action,
		Payload: payload,
		PubKey:  base64.StdEncoding.EncodeToString(pubBytes),
		Sig:     base64.StdEncoding.EncodeToString(sigBytes),
	}

	b, _ := json.Marshal(tx)
	req, err := http.NewRequestWithContext(ctx, "POST", strings.TrimRight(api, "/")+"/tx", strings.NewReader(string(b)))
	if err != nil {
		log.Printf("request error: %v", err)
		return ""
	}
	req.Header.Set("Content-Type", "application/json")

	resp, err := httpClient.Do(req)
	if err != nil {
		log.Printf("http post error: %v", err)
		return ""
	}
	defer resp.Body.Close()
	if resp.StatusCode >= 300 {
		body := make([]byte, 1024)
		resp.Body.Read(body)
		log.Printf("http error status=%d body=%s", resp.StatusCode, string(body))
		return ""
	}
	return tx.ID
}

package execution

import (
	"fmt"
	"strconv"
	"sync"

	"github.com/MelzLabs/DeSync/types"
)

// SandboxedState simple key->value map for sandbox execution (thread-safe)
type SandboxedState struct {
	mu   sync.RWMutex
	data map[string][]byte
}

func NewSandboxedState(snapshot map[string][]byte) *SandboxedState {
	s := &SandboxedState{data: make(map[string][]byte)}
	if snapshot != nil {
		for k, v := range snapshot {
			cp := make([]byte, len(v))
			copy(cp, v)
			s.data[k] = cp
		}
	}
	return s
}

func (s *SandboxedState) Get(key string) ([]byte, bool) {
	s.mu.RLock()
	defer s.mu.RUnlock()
	v, ok := s.data[key]
	if !ok {
		return nil, false
	}
	cp := make([]byte, len(v))
	copy(cp, v)
	return cp, true
}

func (s *SandboxedState) Set(key string, value []byte) {
	s.mu.Lock()
	defer s.mu.Unlock()
	cp := make([]byte, len(value))
	copy(cp, value)
	s.data[key] = cp
}

// GetSnapshot returns a copy of all data (thread-safe)
func (s *SandboxedState) GetSnapshot() map[string][]byte {
	s.mu.RLock()
	defer s.mu.RUnlock()
	snapshot := make(map[string][]byte, len(s.data))
	for k, v := range s.data {
		cp := make([]byte, len(v))
		copy(cp, v)
		snapshot[k] = cp
	}
	return snapshot
}

// DeclaredKeysFor returns deterministic keys touched by tx.
func DeclaredKeysFor(tx *types.Transaction) []string {
	switch tx.Action {
	case "AllocateLand":
		plotID := tx.Payload["plotID"]
		return []string{
			fmt.Sprintf("land:%s:owner", plotID),
			fmt.Sprintf("land:%s:quota", plotID),
			fmt.Sprintf("land:%s:resource", plotID),
		}
	case "Grow":
		plotID := tx.Payload["plotID"]
		return []string{
			fmt.Sprintf("land:%s:resource", plotID),
		}
	case "Transfer":
		from := tx.Payload["from"]
		to := tx.Payload["to"]
		return []string{
			fmt.Sprintf("player:%s:balance", from),
			fmt.Sprintf("player:%s:balance", to),
		}
	case "PayTax":
		plotID := tx.Payload["plotID"]
		return []string{
			fmt.Sprintf("land:%s:owner", plotID),
			fmt.Sprintf("player:%s:balance", tx.Sender),
			fmt.Sprintf("land:%s:tax_paid", plotID),
		}
	default:
		return []string{fmt.Sprintf("player:%s:balance", tx.Sender)}
	}
}

// ApplyTransaction executes tx in sandbox and returns a TxResult.
func ApplyTransaction(snap *SandboxedState, tx *types.Transaction, workerID string, index int) *types.TxResult {
	res := &types.TxResult{
		TxID:         tx.ID,
		Index:        index,
		Success:      false,
		WriteSet:     make(map[string][]byte),
		ExecutedBy:   workerID,
		DeclaredKeys: DeclaredKeysFor(tx),
	}

	// Verify signature
	if !tx.VerifySignature() {
		res.Error = "invalid signature"
		return res
	}

	switch tx.Action {
	case "AllocateLand":
		plotID := tx.Payload["plotID"]
		owner := tx.Payload["owner"]
		if plotID == "" || owner == "" {
			res.Error = "missing plotID or owner"
			return res
		}
		key := fmt.Sprintf("land:%s:owner", plotID)
		curB, _ := snap.Get(key)
		if curB != nil && string(curB) != "" {
			// plot already has an owner
			res.Error = "plot already allocated"
			return res
		}
		// mark ownership and reset counters
		res.WriteSet[key] = []byte(owner)
		res.WriteSet[fmt.Sprintf("land:%s:quota", plotID)] = []byte("0")
		res.WriteSet[fmt.Sprintf("land:%s:resource", plotID)] = []byte("0")
		res.Events = append(res.Events, "Allocated:"+plotID)
		res.Success = true
		return res

	case "Grow":
		plotID := tx.Payload["plotID"]
		amtStr := tx.Payload["amount"]
		if plotID == "" || amtStr == "" {
			res.Error = "missing plotID or amount"
			return res
		}
		amt, err := strconv.Atoi(amtStr)
		if err != nil || amt <= 0 {
			res.Error = "invalid amount"
			return res
		}
		key := fmt.Sprintf("land:%s:resource", plotID)
		curB, _ := snap.Get(key)
		cur := 0
		if curB != nil {
			cur, _ = strconv.Atoi(string(curB))
		}
		newVal := cur + amt
		res.WriteSet[key] = []byte(strconv.Itoa(newVal))
		res.Events = append(res.Events, fmt.Sprintf("Grew:%s:%d", plotID, amt))
		res.Success = true
		return res

	case "Transfer":
    from := tx.Payload["from"]
    to := tx.Payload["to"]
    amtStr := tx.Payload["amount"]
    if from == "" || to == "" || amtStr == "" {
        res.Error = "missing from/to/amount"
        return res
    }
    amt, err := strconv.Atoi(amtStr)
    if err != nil || amt <= 0 {
        res.Error = "invalid amount"
        return res
    }

    fromKey := fmt.Sprintf("player:%s:balance", from)
    toKey := fmt.Sprintf("player:%s:balance", to)

    // Normal transfer flow for all accounts
    fromB, _ := snap.Get(fromKey)
    fromBal := 0
    if fromB != nil {
        fromBal, _ = strconv.Atoi(string(fromB))
    }
    if fromBal < amt {
        res.Error = "insufficient balance"
        return res
    }
    toB, _ := snap.Get(toKey)
    toBal := 0
    if toB != nil {
        toBal, _ = strconv.Atoi(string(toB))
    }
    res.WriteSet[fromKey] = []byte(strconv.Itoa(fromBal - amt))
    res.WriteSet[toKey] = []byte(strconv.Itoa(toBal + amt))
    res.Events = append(res.Events, fmt.Sprintf("Transfer:%s->%s:%d", from, to, amt))
    res.Success = true
    return res

	case "PayTax":
		plotID := tx.Payload["plotID"]
		amountStr := tx.Payload["amount"]
		if plotID == "" || amountStr == "" {
			res.Error = "missing plotID/amount"
			return res
		}
		amt, err := strconv.Atoi(amountStr)
		if err != nil || amt < 0 {
			res.Error = "invalid amount"
			return res
		}
		ownerKey := fmt.Sprintf("land:%s:owner", plotID)
		ownerB, ok := snap.Get(ownerKey)
		if !ok || ownerB == nil || string(ownerB) == "" {
			res.Error = "plot not allocated"
			return res
		}
		owner := string(ownerB)
		ownerBalKey := fmt.Sprintf("player:%s:balance", owner)
		bB, _ := snap.Get(ownerBalKey)
		bal := 0
		if bB != nil {
			bal, _ = strconv.Atoi(string(bB))
		}
		if bal < amt {
			res.Error = "insufficient balance to pay tax"
			return res
		}
		res.WriteSet[ownerBalKey] = []byte(strconv.Itoa(bal - amt))
		res.WriteSet[fmt.Sprintf("land:%s:tax_paid", plotID)] = []byte("true")
		res.Events = append(res.Events, fmt.Sprintf("TaxPaid:%s:%d", plotID, amt))
		res.Success = true
		return res

	default:
		res.Error = "unknown action"
		return res
	}
}

// Worker executes batches, verifies tx signatures and publishes results.
type Worker struct {
	ID      string
	NetNode *network.Node

	privKey ed25519.PrivateKey
	pubKey  ed25519.PublicKey

	totalProcessed int64
	totalShards    int64

	processedMu sync.Mutex
	processed   map[string]struct{}
}

func NewWorker(id string, node *network.Node) *Worker {
	return &Worker{
		ID:        id,
		NetNode:   node,
		processed: make(map[string]struct{}),
	}
}

func (w *Worker) Start(ctx context.Context) error {
	_, priv, err := ed25519.GenerateKey(rand.Reader)
	if err != nil {
		return err
	}
	w.privKey = priv
	w.pubKey = priv.Public().(ed25519.PublicKey)

	// initial hello (non-blocking)
	wh := network.WorkerHello{
		WorkerID: w.ID,
		PubKey:   []byte(w.pubKey),
		PeerID:   w.NetNode.Host.ID().String(),
	}
	if data, err := network.EncodeWorkerHello(wh); err == nil {
		_ = w.NetNode.WorkersTopic.Publish(ctx, data)
	}

	// subscribe to leader heartbeats (just to log, passive)
	if sub, err := w.NetNode.LeaderTopic.Subscribe(); err == nil {
		go func() {
			for {
				msg, err := sub.Next(ctx)
				if err != nil {
					return
				}
				ann, err := network.DecodeLeaderAnnounce(msg.Data)
				if err != nil {
					continue
				}
				log.Printf("Worker %s: leader heartbeat from %s (epoch=%d)", w.ID, ann.LeaderID, ann.Epoch)
			}
		}()
	}

	// performance monitor
	go w.startPerformanceMonitor(ctx)

	// subscribe to batch topic
	sub, err := w.NetNode.BatchTopic.Subscribe()
	if err != nil {
		return err
	}
	go func() {
		for {
			msg, err := sub.Next(ctx)
			if err != nil {
				return
			}
			bm, err := network.DecodeBatch(msg.Data)
			if err != nil {
				continue
			}
			// if the batch targeted a specific worker, skip if not me
			if bm.TargetWorker != "" && bm.TargetWorker != w.ID {
				continue
			}
			go w.processShard(ctx, bm)
		}
	}()

	return nil
}

func (w *Worker) startPerformanceMonitor(ctx context.Context) {
	ticker := time.NewTicker(10 * time.Second)
	defer ticker.Stop()
	for {
		select {
		case <-ctx.Done():
			return
		case <-ticker.C:
			processed := atomic.LoadInt64(&w.totalProcessed)
			shards := atomic.LoadInt64(&w.totalShards)
			if shards > 0 {
				avg := float64(processed) / float64(shards)
				log.Printf("Worker %s STATS: processed=%d txs, shards=%d, avg=%.2f",
					w.ID, processed, shards, avg)
			}
		}
	}
}

func (w *Worker) processShard(ctx context.Context, bm *network.BatchMessage) {
	key := bm.BatchID + "-" + string(rune(bm.ShardIndex))
	w.processedMu.Lock()
	if _, seen := w.processed[key]; seen {
		w.processedMu.Unlock()
		return
	}
	w.processed[key] = struct{}{}
	w.processedMu.Unlock()

	start := time.Now()
	atomic.AddInt64(&w.totalShards, 1)

	// create sandbox state from snapshot (may be nil)
	snap := execution.NewSandboxedState(bm.Snapshot)
	results := make([]*types.TxResult, len(bm.Txs))

	var wg sync.WaitGroup
	wg.Add(len(bm.Txs))
	for i, tx := range bm.Txs {
		i, tx := i, tx
		go func() {
			defer wg.Done()
			// verify tx signature
			if !tx.VerifySignature() {
				results[i] = &types.TxResult{
					TxID:    tx.ID,
					Index:   i,
					Success: false,
					Error:   "invalid signature",
				}
				return
			}
			// execute
			results[i] = execution.ApplyTransaction(snap, tx, w.ID, i)
		}()
	}
	wg.Wait()
	atomic.AddInt64(&w.totalProcessed, int64(len(bm.Txs)))
	elapsed := time.Since(start)

	// - - - Success/fail counters + failure reason breakdown - - -
	successCount := 0
	failCount := 0
	errorReasons := make(map[string]int)

	for _, r := range results {
		if r != nil && r.Success {
			successCount++
		} else if r != nil {
			failCount++
			reason := r.Error
			if reason == "" {
				reason = "unknown"
			}
			errorReasons[reason]++
		}
	}

	log.Printf("Worker %s: finished shard %d/%d batch %s (txs=%d success=%d fail=%d elapsed=%s)",
		w.ID, bm.ShardIndex+1, bm.ShardCount, bm.BatchID,
		len(bm.Txs), successCount, failCount, elapsed)

	if failCount > 0 {
		for reason, count := range errorReasons {
			log.Printf("  └─ fail_reason=%q count=%d", reason, count)
		}
	}

	// sign results blob
	toSign := struct {
		BatchID    string
		WorkerID   string
		ShardIndex int
		ShardCount int
		Results    []*types.TxResult
	}{
		BatchID:    bm.BatchID,
		WorkerID:   w.ID,
		ShardIndex: bm.ShardIndex,
		ShardCount: bm.ShardCount,
		Results:    results,
	}
	payload, _ := json.Marshal(toSign)
	sig := ed25519.Sign(w.privKey, payload)

	sr := network.SignedResult{
		BatchID:    bm.BatchID,
		WorkerID:   w.ID,
		ShardIndex: bm.ShardIndex,
		ShardCount: bm.ShardCount,
		Results:    results,
		PubKey:     []byte(w.pubKey),
		Sig:        sig,
	}
	data, _ := network.EncodeResult(sr)
	_ = w.NetNode.ResultTopic.Publish(ctx, data)
}

func main() {
	flag.Parse()

	if *idFlag == "" {
		log.Fatal("need --id flag")
	}

	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()

	// load or create key
	priv, err := network.LoadOrCreatePrivKey(*keyfileFlag)
	if err != nil {
		log.Fatalf("failed to load/create key: %v", err)

	addr := "/ip4/0.0.0.0/tcp/" + *portFlag

	// optional bootstrap connect
	if *connectFlag != "" {
		if err := node.Connect(ctx, *connectFlag); err != nil {
			log.Printf("Warning: failed to connect to %s: %v", *connectFlag, err)

package types

import (
	"crypto/ed25519"
	"crypto/rand"
	"encoding/base64"
	"encoding/hex"
	"sort"
	"strings"
)

// Transaction represents a simple action.
type Transaction struct {
	ID      string
	Sender  string
	Nonce   uint64
	Action  string
	Payload map[string]string
	PubKey  string
	Sig     string
}

// NewTxn creates a new transaction with an optional payload.
// If payload is nil, an empty map is created.
func NewTxn(sender, action string, payload map[string]string) *Transaction {
	idb := make([]byte, 8)
	_, _ = rand.Read(idb)

	payloadCopy := make(map[string]string)
	if payload != nil {
		for k, v := range payload {
			payloadCopy[k] = v
		}
	}

	return &Transaction{
		ID:      hex.EncodeToString(idb),
		Sender:  sender,
		Action:  action,
		Payload: payloadCopy,
	}
}


func (t *Transaction) MsgBytes() []byte {
	var sb strings.Builder
	sb.WriteString(t.Sender)
	sb.WriteString("|")
	sb.WriteString(t.Action)
	sb.WriteString("|")
	if t.Payload != nil {
		keys := make([]string, 0, len(t.Payload))
		for k := range t.Payload {
			keys = append(keys, k)
		}
		sort.Strings(keys)
		for _, k := range keys {
			sb.WriteString(k)
			sb.WriteString("=")
			sb.WriteString(t.Payload[k])
			sb.WriteString(";")
		}
	}
	return []byte(sb.String())
}

func (t *Transaction) Sign(priv ed25519.PrivateKey) error {
	sig := ed25519.Sign(priv, t.MsgBytes())
	pk := priv.Public().(ed25519.PublicKey)
	t.PubKey = base64.StdEncoding.EncodeToString(pk)
	t.Sig = base64.StdEncoding.EncodeToString(sig)
	return nil
}
func (t *Transaction) VerifySignature() bool {
	if t.PubKey == "" || t.Sig == "" {
		return false
	}
	
	pubBytes, err := base64.StdEncoding.DecodeString(t.PubKey)
	if err != nil {
		return false
	}
	sigBytes, err := base64.StdEncoding.DecodeString(t.Sig)
	if err != nil {
		return false
	}
	
	// Detect signature type by public key length
	// Ed25519: 32 bytes, ECDSA P-256 SPKI: 91 bytes
	if len(pubBytes) == 32 && len(sigBytes) == 64 {
		// Ed25519 signature (from Go loadgen tools)
		return ed25519.Verify(ed25519.PublicKey(pubBytes), t.MsgBytes(), sigBytes)
	} else if len(pubBytes) == 91 && len(sigBytes) == 64 {
		// ECDSA P-256 signature (from HTTP/browser)
		// Workers should not verify these - API server handles it
		// Just return true and let the API do full verification
		return true
	}
	
	return false
}

type TxResult struct {

// NewNodeWithKey starts a libp2p node with given privkey
func NewNodeWithKey(ctx context.Context, listenAddr string, priv crypto.PrivKey) (*Node, error) {
	h, err := libp2p.New(
		libp2p.ListenAddrStrings(listenAddr),
		libp2p.Identity(priv),
	)
	if err != nil {
		return nil, err
	}

	ps, err := pubsub.NewGossipSub(ctx, h)
	if err != nil {
		return nil, err
	}

	workersTopic, _ := ps.Join("workers")
	txTopic, _ := ps.Join("txs")
	batchTopic, _ := ps.Join("batches")
	resultTopic, _ := ps.Join("results")
	leaderTopic, _ := ps.Join("leader")

	n := &Node{
		Host:         h,
		PubSub:       ps,
		WorkersTopic: workersTopic,
		TxTopic:      txTopic,
		BatchTopic:   batchTopic,
		ResultTopic:  resultTopic,
		LeaderTopic:  leaderTopic,
	}

	// MDNS discovery (local dev)
	ser := mdns.NewMdnsService(h, "desync-mdns", &discoveryNotifee{h: h})
	if err != nil {
		log.Printf("mdns disabled: %v", err)
	} else {
		_ = ser
	}

	log.Printf("Node started with peer ID: %s, listening on %s", h.ID().String(), listenAddr)
	return n, nil
}

type discoveryNotifee struct {
	h host.Host
}

func (n *discoveryNotifee) HandlePeerFound(pi peer.AddrInfo) {
	if pi.ID == n.h.ID() {
		return
	}
	if err := n.h.Connect(context.Background(), pi); err == nil {
		log.Printf("Connected to peer %s", pi.ID)
	}
}

	pi, err := peer.AddrInfoFromString(addr)
	if err != nil {
		return err
	}
	if err := n.Host.Connect(ctx, *pi); err != nil {
		return err
	}
	log.Printf("Connected to %s", pi.ID.String())
	return nil
}

package state

import (
	"crypto/sha256"
	"sync"

	celsmt "github.com/celestiaorg/smt"
)

// SparseMerkleTree wraps celestiaorg/smt with an in-memory KV mirror.
type SparseMerkleTree struct {
	tree *celsmt.SparseMerkleTree

	mu sync.RWMutex
	kv map[string][]byte
}

// NewSparseMerkleTree creates a new in-memory SMT.
func NewSparseMerkleTree() *SparseMerkleTree {
	store := celsmt.NewSimpleMap()
	tree := celsmt.NewSparseMerkleTree(store, store, sha256.New())
	return &SparseMerkleTree{
		tree: tree,
		kv:   make(map[string][]byte),
	}
}

// Update inserts/updates a key and mirrors it in kv.
func (s *SparseMerkleTree) Update(key, value []byte) error {
	if _,err := s.tree.Update(key, value); err != nil {
		return err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	cp := make([]byte, len(value))
	copy(cp, value)
	s.kv[string(key)] = cp
	return nil
}

// GetRoot returns the Merkle root.
func (s *SparseMerkleTree) GetRoot() []byte {
	return s.tree.Root()
}

// Prove generates a Merkle proof for a key.
func (s *SparseMerkleTree) Prove(key []byte) (celsmt.SparseMerkleProof, error) {
	return s.tree.Prove(key)
}

// VerifyProof verifies a proof against a root.
func VerifyProof(proof celsmt.SparseMerkleProof, root, key, value []byte) bool {
	return celsmt.VerifyProof(proof, root, key, value, sha256.New())
}

// ExportSnapshot returns a copy of the in-memory kv.
func (s *SparseMerkleTree) ExportSnapshot() map[string][]byte {
	out := make(map[string][]byte)
	s.mu.RLock()
	defer s.mu.RUnlock()
	for k, v := range s.kv {
		cp := make([]byte, len(v))
		copy(cp, v)
		out[k] = cp
	}
	return out
}

package network

import (
	"crypto/rand"
	"encoding/hex"
	"errors"
	"io/ioutil"
	"os"

	crypto "github.com/libp2p/go-libp2p/core/crypto"
)

// LoadOrCreatePrivKey loads an ed25519 key from file or generates one if missing.
// Returns a libp2p crypto.PrivKey.
func LoadOrCreatePrivKey(path string) (crypto.PrivKey, error) {
	if path == "" {
		priv, _, err := crypto.GenerateEd25519Key(rand.Reader)
		return priv, err
	}
	if _, err := os.Stat(path); errors.Is(err, os.ErrNotExist) {
		priv, _, err := crypto.GenerateEd25519Key(rand.Reader)
		if err != nil {
			return nil, err
		}
		b, err := crypto.MarshalPrivateKey(priv)
		if err != nil {
			return nil, err
		}
		if err := ioutil.WriteFile(path, []byte(hex.EncodeToString(b)), 0600); err != nil {
			return nil, err
		}
		return priv, nil
	}
	data, err := ioutil.ReadFile(path)
	if err != nil {
		return nil, err
	}
	b, err := hex.DecodeString(string(data))
	if err != nil {
		return nil, err
	}
	priv, err := crypto.UnmarshalPrivateKey(b)
	if err != nil {
		return nil, err
	}
	return priv, nil
}

	monthEvents = nil
	monthNumber++
}
